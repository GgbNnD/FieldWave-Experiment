\section{实验原理}
\begin{figure}[H]
  \centering
  \includegraphics[width=0.8\textwidth]{原理图.png}
  \caption{微带线负载特性矢网网络分析原理图} 
\end{figure}

考虑一段特性阻抗为$Z_0$的传输线，一端接信号源，另一端则接上负载。
假设此传输线无耗，且传输线$\gamma = j \beta$，则传输线上电压及电流可用
下列二式表示：
\begin{align*}
  U(z) = U^+e^{-\beta z} + U^-e^{\beta z} \\
  I(z) = I^+e^{-\beta z} - I^-e^{\beta z}
\end{align*}

\subsection{负载端(z = 0)处情况}
电压及电流为\par
\begin{align*}
  U = U_L = U^+ + U^- \\
  I = I_L = I^+ - I^-
\end{align*}
而$Z_0I^+ = U^+,Z_0I^- = U^-$，公式可改写成\par
\begin{align*}
  I_L = \frac{1}{Z_0}(U^+-U^-)
\end{align*}
可得负载阻抗为\par
\begin{align*}
  Z_L = \frac{U_L}{I_L} = Z_0\frac{U^+ + U^-}{U^+ - U^-}
\end{align*}
定义归一化负载阻抗为\par
\begin{align*}
  z_L = Z_L = \frac{Z_L}{Z_0} = \frac{1+\Gamma_L}{1-\Gamma_L}
\end{align*}
其中定义$\Gamma_L$为负载端的电压反射系数\par
\begin{align*}
  \Gamma_L = \frac{U^-}{U^+} = \frac{Z_L - Z_0}{Z_L + Z_0} = |\Gamma_L|e^{j\phi_L}
\end{align*}
当$Z_L = Z_0$或为无限长传输线时，$\Gamma_L = 0$，无反射波，是行波状态或匹配状态。\par
当$Z_L$为纯电抗元件或处于开路或者短路状态时，$|\Gamma_L| = 1$，全反射，是驻波状态。\par
当$Z_L$为其他值时，$|\Gamma_L| <= 1$，为行波驻波状态。\par
线上任意点的反射系数为\par
\begin{align*}
  \Gamma_L = |\Gamma_L|e^{j\phi_L-j2\beta z}
\end{align*}
定义驻波比VSWR和回波损耗RL为\par
\begin{align*}
  VSWR = \frac{1+|\Gamma_L|}{1-|\Gamma_L|}\\
  RL = -20\log_{10}|\Gamma_L|
\end{align*}

\subsection{输入端(z = -l)处情况}
反射系数$\Gamma(z)$应改成\par
\begin{align*}
  \Gamma(L) = \frac{U^-e^{-j\beta_L}}{U^+e^{j\phi\beta_L}} = \frac{U^-}{U^+}e^{-j2\beta_L} = \Gamma_L e^{-j2\beta_L}
\end{align*}
输入阻抗为\par
\begin{align*}
  Z_{in} = Z_0\frac{Z_L+jZ_0\tan(\beta L)}{Z_0+jZ_L\tan(\beta L)}
\end{align*}
由上式可知\par
\begin{itemize}
  \item 当$L \rightarrow \infty$时，$Z_{in} \rightarrow Z_0$
  \item 当$L = \frac{\lambda}{2}$时，$Z_{in} = Z_L$
  \item 当$L = \frac{\lambda}{4}$时，$Z_{in} = \frac{Z_0^2}{Z_L}$
\end{itemize}
