\section{实验原理}
\begin{figure}[H]
  \centering
  \includegraphics[width=0.8\textwidth]{原理图.png}
  \caption{微带线负载特性矢网网络分析原理图} 
\end{figure}

考虑一段特性阻抗为$Z_0$的传输线，一端接信号源，另一端则接上负载。
假设此传输线无耗，且传输线$\gamma = j \beta$，则传输线上电压及电流可用
下列二式表示：
\begin{align*}
  U(z) = U^+e^{-\beta z} + U^-e^{\beta z} \\
  I(z) = I^+e^{-\beta z} - I^-e^{\beta z}
\end{align*}

\subsection{负载端(z = 0)处情况}
电压及电流为\par
\begin{align*}
  U = U_L = U^+ + U^- \\
  I = I_L = I^+ - I^-
\end{align*}
而$Z_0I^+ = U^+,Z_0I^- = U^-$，公式可改写成\par
\begin{align*}
  I_L = \frac{1}{Z_0}(U^+-U^-)
\end{align*}
可得负载阻抗为\par
\begin{align*}
  Z_L = \frac{U_L}{I_L} = Z_0\frac{U^+ + U^-}{U^+ - U^-}
\end{align*}
定义归一化负载阻抗为\par
\begin{align*}
  z_L = Z_L = \frac{Z_L}{Z_0} = \frac{1+\Gamma_L}{1-\Gamma_L}
\end{align*}
其中定义$\Gamma_L$为负载端的电压反射系数\par
\begin{align*}
  \Gamma_L = \frac{U^-}{U^+} = \frac{Z_L - Z_0}{Z_L + Z_0} = |\Gamma_L|e^{j\phi_L}
\end{align*}
当$Z_L = Z_0$或为无限长传输线时，$\Gamma_L = 0$，无反射波，是行波状态或匹配状态。\par
当$Z_L$为纯电抗元件或处于开路或者短路状态时，$|\Gamma_L| = 1$，全反射，是驻波状态。\par
当$Z_L$为其他值时，$|\Gamma_L| <= 1$，为行波驻波状态。\par
线上任意点的反射系数为\par
\begin{align*}
  \Gamma_L = |\Gamma_L|e^{j\phi_L-j2\beta z}
\end{align*}
定义驻波比VSWR和回波损耗RL为\par
\begin{align*}
  VSWR = \frac{1+|\Gamma_L|}{1-|\Gamma_L|}\\
  RL = -20\log_{10}|\Gamma_L|
\end{align*}

\subsection{输入端(z = -l)处情况}
反射系数$\Gamma(z)$应改成\par
\begin{align*}
  \Gamma(L) = \frac{U^-e^{-j\beta_L}}{U^+e^{j\phi\beta_L}} = \frac{U^-}{U^+}e^{-j2\beta_L} = \Gamma_L e^{-j2\beta_L}
\end{align*}
输入阻抗为\par
\begin{align*}
  Z_{in} = Z_0\frac{Z_L+jZ_0\tan(\beta L)}{Z_0+jZ_L\tan(\beta L)}
\end{align*}
由上式可知\par
\begin{itemize}
  \item 当$L \rightarrow \infty$时，$Z_{in} \rightarrow Z_0$
  \item 当$L = \frac{\lambda}{2}$时，$Z_{in} = Z_L$
  \item 当$L = \frac{\lambda}{4}$时，$Z_{in} = \frac{Z_0^2}{Z_L}$
\end{itemize}

\section{思考题}
\subsection{(1)什么是s参数}
S参数（Scattering Parameters，散射参数）是高频和微波电路中用于描述线性网络输入输出特性的重要工具。
S参数描述的是入射波与反射波之间的关系，通过测量网络端口的电压波（而非直接电压或电流）来表征网络的传输和反射特性。
$S_{ii}$表示端口$i$的反射系数，$S_{ij}$表示端口$i$到端口$j$的传输系数。\\

\subsection{(2)如果不校准，直接接入射频电缆和电路模块测量会对结果有什么影响}
如果不校准，直接接入射频电缆和电路模块测量，结果可能出现显著误差。
射频电缆和连接器会引入插入损耗和相位延迟，导致测得的$S_{21}$偏小、相位失真。\\
连接器或电缆阻抗不连续会引起多次反射，导致$S_{11}$和$S_{22}$出现虚假峰值。\\

\subsection{(3)如何测量转接头对测试曲线的影响}
接上转接头后，用分析仪测量完全匹配阻抗的史密斯圆图，所得曲线与匹配点之间的偏移即为转接头产生的误差。

\subsection{(4)计算半波长微带线的长度和宽度}
由$L = \frac{\lambda}{2} = \frac{c}{2f},f = 2.5\times 10^9 Hz$，得$L = 6 cm$\par
由$H = 0.765 \times 10^{-3}m,\varepsilon_r = 4.6,T = 0.035\times 10^{-3}m,Z_0 = 50\Omega$
\begin{align*}
  Z_0 = \frac{87}{\sqrt{\varepsilon_r + 1.41}} \ln \left( \frac{5.98H}{0.8W + T} \right) \\
\end{align*}
得$W = 1.33mm$
